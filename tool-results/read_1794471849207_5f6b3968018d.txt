    29→const statusBadge = (p: Product) => {
    30→  if (p.status === 'DRAFT') return { label: STATUS_LABELS.DRAFT, cls: 'bg-slate-100 text-slate-600 dark:bg-slate-900 dark:text-slate-300' }
    31→  if (p.status === 'OUT' || p.stock === 0) return { label: STATUS_LABELS.OUT, cls: 'bg-red-100 text-red-700 dark:bg-red-950 dark:text-red-300' }
    32→  if (p.status === 'ARCHIVED') return { label: STATUS_LABELS.ARCHIVED, cls: 'bg-amber-100 text-amber-700 dark:bg-amber-950 dark:text-amber-300' }
    33→  if (p.stock <= p.lowStockThreshold) return { label: 'کم‌موجودی', cls: 'bg-amber-100 text-amber-700 dark:bg-amber-950 dark:text-amber-300' }
    34→  return { label: STATUS_LABELS.PUBLISHED, cls: 'bg-emerald-100 text-emerald-700 dark:bg-emerald-950 dark:text-emerald-300' }
    35→}
